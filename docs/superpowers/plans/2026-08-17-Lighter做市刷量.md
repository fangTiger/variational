### ⚠️ 两个最危险的 API 契约

**一、`base_amount` 和 `price` 是整数，不是小数。** BTC 市场实测：

| 字段 | 精度 | 换算 | 例 |
|---|---|---|---|
| `base_amount` | `size_decimals = 5` | BTC × 100000 | 0.00317 BTC → `317` |
| `price` | `price_decimals = 1` | USD × 10 | 63400.0 → `634000` |

**换算错一位就是 10 倍或 1/10 的仓位。** 这是本项目最高风险点，Task 2 用大量边界测试锁死。

**二、错误在返回值第三个元素里，不抛异常。**

```python
tx, resp, err = await signer.create_order(...)
if err is not None:
    raise LighterOrderError(err)   # 必须显式检查，否则失败被静默吞掉
```

忘记检查 `err` 会让「下单失败」看起来像「下单成功」，引擎据此更新内部状态，后果是账实不符。

```python
"""Lighter 下单参数的整数换算。

Lighter 的 create_order 接收整数而非小数：
    base_amount = BTC 数量 × 10**size_decimals
    price       = USD 价格 × 10**price_decimals

BTC 实测 size_decimals=5、price_decimals=1，即
    0.00317 BTC → 317
    63400.0 USD → 634000

换算错一位就是 10 倍仓位，因此这里全部用 Decimal、一律向下取整、
并对零值/负值/低于精度的输入直接报错——宁可下不出单，不能下错单。
"""

from __future__ import annotations

from decimal import Decimal


def _check(value: Decimal, what: str) -> None:
    if not isinstance(value, Decimal):
        raise TypeError(f"{what}必须是 Decimal，收到 {type(value).__name__}；浮点会引入精度误差")
    if value <= 0:
        raise ValueError(f"{what}必须为正，收到 {value}；方向由 is_ask 表达")


def to_base_amount(amount: Decimal, size_decimals: int) -> int:
    """BTC 数量 → 整数。向下取整：向上会让实际下单量超过引擎预期。"""
    _check(amount, "数量")
    scaled = int(amount * (10 ** size_decimals))
    if scaled == 0:
        raise ValueError(f"数量 {amount} 小于最小精度单位（10^-{size_decimals}）")
    return scaled


def to_price(price: Decimal, price_decimals: int) -> int:
    """USD 价格 → 整数。同样向下取整。"""
    _check(price, "价格")
    scaled = int(price * (10 ** price_decimals))
    if scaled == 0:
        raise ValueError(f"价格 {price} 小于最小精度单位（10^-{price_decimals}）")
    return scaled


def from_base_amount(scaled: int, size_decimals: int) -> Decimal:
    """整数 → BTC 数量。"""
    return Decimal(scaled) / (10 ** size_decimals)


def from_price(scaled: int, price_decimals: int) -> Decimal:
    """整数 → USD 价格。"""
    return Decimal(scaled) / (10 ** price_decimals)
```

```python
    async def _load_market_meta(self, market: str) -> dict:
        """缓存市场精度参数。换算依赖它，绝不能猜。"""
        if getattr(self, "_market_meta", None) is None:
            self._market_meta = {}
        if market not in self._market_meta:
            data = await self._get_json("/api/v1/orderBooks")
            self._raise_api_error(data, context="市场列表查询")
            for book in data.get("order_books") or []:
                if str(book.get("symbol", "")).upper() == market.upper():
                    self._market_meta[market] = {
                        "market_id": int(book["market_id"]),
                        "size_decimals": int(book["supported_size_decimals"]),
                        "price_decimals": int(book["supported_price_decimals"]),
                    }
                    break
            else:
                raise KeyError(f"Lighter 没有标的 {market}")
        return self._market_meta[market]

    def _require_signer(self):
        if self._signer is None:
            import lighter

            self._signer = lighter.SignerClient(
```

```python
"""对冲机器人存活检查。

做市机器人在 Lighter 上不断产生库存，全靠对冲机器人在 Extended 上抵消。
若对冲挂了而做市继续跑，库存会一路累积成裸仓——这是本系统最危险的失败模式。

因此做市方每轮都要问一句「对冲还活着吗」，答案不确定时一律按不活处理：
宁可停下不赚钱，不能裸奔。
"""

from __future__ import annotations

import json
import time
from dataclasses import dataclass
from pathlib import Path

_DEFAULT_INTERVAL = 30.0


@dataclass(frozen=True)
class HedgeLiveness:
    """对冲存活判定结果。reason 会写进日志和告警，必须可读。"""

    alive: bool
    reason: str


def _last_row(path: Path) -> dict | None:
    try:
        lines = Path(path).read_text(encoding="utf-8").splitlines()
    except Exception:  # noqa: BLE001 缺失或损坏都按不可信处理
        return None
    for line in reversed(lines):
        if not line.strip():
            continue
        try:
            row = json.loads(line)
        except Exception:  # noqa: BLE001
            continue
        if isinstance(row, dict):
            return row
    return None


def check_hedge_alive(snapshot_path, *, now: float | None = None) -> HedgeLiveness:
    """判断对冲机器人是否在正常工作。任何不确定都返回 alive=False。"""
    now = time.time() if now is None else now
    row = _last_row(snapshot_path)
    if row is None:
        return HedgeLiveness(False, "读不到对冲心跳文件，对冲可能从未启动")

    ts = row.get("ts")
    try:
        age = now - float(ts)
    except (TypeError, ValueError):
        return HedgeLiveness(False, "对冲心跳缺少可用的时间戳")

    try:
        interval = float(row.get("interval") or _DEFAULT_INTERVAL)
    except (TypeError, ValueError):
        interval = _DEFAULT_INTERVAL
    if interval <= 0:
        interval = _DEFAULT_INTERVAL

    if age > 3 * interval:
        return HedgeLiveness(False, f"对冲心跳陈旧（{age:.0f} 秒未更新）")

    if not row.get("primary_read_ok"):
        return HedgeLiveness(False, "对冲进程读不到 Lighter 持仓，等于没在对冲")

    return HedgeLiveness(True, "对冲正常")
```

```python
@dataclass(frozen=True)
class HedgeFillResult:
    """一次对冲下单的结果。note 记录异常路径，供排查。"""

    used_taker: bool
    note: str = ""


async def maker_first_hedge(
    adapter,
    market: str,
    target_delta: Decimal,
    limit_price: Decimal,
    *,
    timeout_s: float = 15.0,
    poll_s: float = 1.0,
) -> HedgeFillResult:
    """先挂 maker，超时未成交则撤单转吃单。

    Extended maker 费率为 0、taker 约 3bps。做市场景下对冲频率很高，
    纯吃单会吃掉网格约 60% 的利润，因此优先走 maker。

    但敞口不能永远悬着——超时必须转吃单。撤单失败时仍然吃单，
    宁可承担「挂单可能也成交」的双倍仓位风险（下一轮再平衡会纠正），
    也不能留一个没被对冲的敞口。
    """
    if target_delta == 0:
        return HedgeFillResult(used_taker=False, note="目标为零，无需下单")

    side = Side.BUY if target_delta > 0 else Side.SELL
    amount = abs(target_delta)
    order = await adapter.place_limit_order(
        market, side, amount, limit_price, post_only=True
    )
    order_id = order.get("id") if isinstance(order, dict) else getattr(order, "id", None)

    deadline = time.monotonic() + timeout_s
    start = (await adapter.get_position(market)).signed_size
    while time.monotonic() < deadline:
        await asyncio.sleep(poll_s)
        current = (await adapter.get_position(market)).signed_size
        if current != start:
            return HedgeFillResult(used_taker=False, note="maker 成交")

    note = ""
    try:
        await adapter.cancel_order(market, order_id)
    except Exception as exc:  # noqa: BLE001 撤单失败也要把敞口对冲上
        note = f"撤单失败（{exc}），仍转吃单；若挂单随后成交会形成双倍仓位，下轮再平衡纠正"
        logger.warning(note)

    await adapter.market_order(market, side, amount)
    return HedgeFillResult(used_taker=True, note=note or "maker 超时，转吃单")
```

```python
"""K 线行情源。

从交易所适配器里独立出来，是为了让「在哪交易」和「用谁的 K 线」解耦：
Lighter 的 K 线接口返回 403，做市引擎必须用 Extended 的 BTC K 线算 band。
两所同一标的，实测基差约 0.08%，用于 ATR/ADX 完全够用。
"""

from __future__ import annotations

from dataclasses import dataclass
from decimal import Decimal
from typing import Protocol


@dataclass(frozen=True)
class Candle:
    """一根 K 线。只保留 band/regime 计算真正用到的字段。"""

    ts: int
    high: Decimal
    low: Decimal
    close: Decimal


class CandleSource(Protocol):
    """K 线来源。实现方决定从哪个交易所取。"""

    async def get_hourly_candles(self, market: str, limit: int) -> list[Candle]:
        ...


class ExtendedCandleSource:
    """从 Extended SDK 取小时 K 线。"""

    def __init__(self, ext) -> None:
        self._ext = ext

    async def get_hourly_candles(self, market: str, limit: int) -> list[Candle]:
        r = await self._ext._client.info.get_candles_history(
            market_name=market,
            candle_type="trades",
            interval="PT1H",
            limit=limit,
        )
        rows = sorted(r.data, key=lambda k: int(k.timestamp))
        return [
            Candle(
                ts=int(k.timestamp),
                high=Decimal(str(k.high)),
                low=Decimal(str(k.low)),
                close=Decimal(str(k.close)),
            )
            for k in rows
        ]
```
