**Goal:** 把对冲腿的下单方式从 IOC 吃单改为「先挂 maker 单、超时转吃单」，把对冲成本从 3bps 降到接近 0，使高频调仓在经济上可持续。

### 为什么做这个

用户计划把 Lighter 日成交量提到 $2 万。对冲腿要镜像同样金额，按当前 IOC 吃单费率：

```
$20,000/天 × 3bps = $6/天 = $180/月
```

两个账户合计 $1,960，即**月化 9.2%、年化 110%** 的纯手续费支出——会成为吃掉本金的主要项。

**Extended 的 maker 费率为 0**（现有网格全部用 post_only 限价单正是靠这一点盈利）。改为 maker 优先后，正常成交路径的手续费降到零。

### ⚠️ 本计划的前一个版本被审查否决，以下是必须修正的三点

上一版设计（做市计划 Task 5）经两位独立审查者审查后发现三个缺陷，本计划已修正，**实施时不要退回旧做法**：

**一、不能用「轮询持仓差值」判断 maker 是否成交。**
旧设计取下单前后的 `get_position().signed_size` 比较。问题：首次读取即竞态（下单后第一次读可能已成交，导致基准值就是成交后的），且**遇到部分成交会误判**——只成交 40% 时差值不为零，函数会返回「已成交」，但剩余 60% 的挂单还活着且订单号已丢弃，既欠对冲又留下孤儿单。

**正确做法：查订单状态。** `ExtendedClient.get_order_by_id(market, order_id)` 返回的对象含 `status` 与 `filled_qty`（实测字段，现有网格引擎就靠它判断成交）。

**二、必须按剩余量吃单，不是按原始量。**
超时转吃单时，若已部分成交，应只吃 `amount - filled_qty`。按原始量吃会造成超额对冲。

**三、撤单失败后必须重读订单再决定。**
旧设计撤单失败仍无条件吃单，理由是「宁可双倍也不能不对冲」。但重读一次订单状态只要一次 API 调用，就能知道它究竟成交了多少，从而精确决定要不要吃、吃多少。无条件吃单是不必要的粗暴。

（另：上一版计划里 `adapter.cancel_order(market, order_id)` 与当时 `ExtendedClient` 的单参数签名冲突，会稳定抛 `TypeError`。该契约已在 `c6a2597` 统一为 `(market, order_id)`，现在两边一致。）

### 挂单价格策略

Extended BTC 盘口价差实测约 $1（如 bid 62965 / ask 62966），**价差内没有插单空间**。因此：

- 对冲要 **SELL** 时，挂在 **best ask**（加入卖一队列）
- 对冲要 **BUY** 时，挂在 **best bid**

挂在对手价会立刻成交从而被 post_only 拒绝，不能那样做。

`get_market_price(market)` 返回 `MarketPrice(bid, ask)`，直接用。

```python
"""maker 优先对冲测试。

核心保证：无论走哪条路径，敞口最终都被精确覆盖——
既不能漏（欠对冲），也不能超（超额对冲）。
"""

from __future__ import annotations

import asyncio
from decimal import Decimal

import pytest

from adapters.base import MarketPrice, Side
from engine.hedge_engine import maker_first_hedge


class _Order:
    """假订单。模拟 ExtendedClient.get_order_by_id 的返回结构。"""

    def __init__(self, order_id, filled=Decimal("0"), status="NEW"):
        self.id = order_id
        self.filled_qty = filled
        self.status = status


class _Adapter:
    """假适配器。签名与 ExtendedClient 一致——本仓库反复踩过签名不符的坑。"""

    def __init__(self, fill_sequence=None, cancel_error=None):
        """fill_sequence: 每次查询订单返回的已成交量序列。"""
        self.limit_orders = []
        self.market_orders = []
        self.cancelled = []
        self._fills = list(fill_sequence or [Decimal("0")])
        self._cancel_error = cancel_error

    async def get_market_price(self, market):
        return MarketPrice(market=market, bid=Decimal("60000"), ask=Decimal("60001"))

    async def place_limit_order(self, market, side, amount, price, *,
                                post_only=True, reduce_only=False):
        self.limit_orders.append(
            {"market": market, "side": side, "amount": amount,
             "price": price, "post_only": post_only}
        )
        return _Order("L1")

    async def get_order_by_id(self, market, order_id):
        filled = self._fills[0] if len(self._fills) == 1 else self._fills.pop(0)
        status = "FILLED" if filled >= self.limit_orders[0]["amount"] else "NEW"
        return _Order(order_id, filled=filled, status=status)

    async def cancel_order(self, market, order_id):
        if self._cancel_error:
            raise RuntimeError(self._cancel_error)
        self.cancelled.append(order_id)

    async def market_order(self, market, side, amount, *, reduce_only=False):
        self.market_orders.append({"side": side, "amount": amount})
        return _Order("M1")


def _run(adapter, delta=Decimal("-1"), **kw):
    return asyncio.run(
        maker_first_hedge(adapter, "BTC-USD", delta, timeout_s=0.05, poll_s=0.01, **kw)
    )


def test_sell_places_maker_at_ask():
    """卖出挂在 best ask。挂在 bid 会立刻成交，被 post_only 拒绝。"""
    a = _Adapter(fill_sequence=[Decimal("1")])
    _run(a, delta=Decimal("-1"))
    assert a.limit_orders[0]["price"] == Decimal("60001")
    assert a.limit_orders[0]["side"] is Side.SELL
    assert a.limit_orders[0]["post_only"] is True


def test_buy_places_maker_at_bid():
    a = _Adapter(fill_sequence=[Decimal("1")])
    _run(a, delta=Decimal("1"))
    assert a.limit_orders[0]["price"] == Decimal("60000")
    assert a.limit_orders[0]["side"] is Side.BUY


def test_full_maker_fill_avoids_taker():
    """maker 全部成交后不得再吃单——那会变成双倍仓位。"""
    a = _Adapter(fill_sequence=[Decimal("1")])
    r = _run(a)
    assert a.market_orders == []
    assert r.used_taker is False
    assert r.filled == Decimal("1")


def test_timeout_cancels_then_takes_full_amount():
    a = _Adapter(fill_sequence=[Decimal("0")])
    r = _run(a)
    assert a.cancelled == ["L1"]
    assert len(a.market_orders) == 1
    assert a.market_orders[0]["amount"] == Decimal("1")
    assert r.used_taker is True


def test_partial_fill_takes_only_remainder():
    """部分成交后只吃差额。按原始量吃会超额对冲。

    这是旧设计最严重的缺陷：用持仓差值判断，只成交 40% 也会被判成
    「已成交」，剩余 60% 既没对冲上，挂单还留在盘口成为孤儿单。
    """
    a = _Adapter(fill_sequence=[Decimal("0.4")])
    r = _run(a)
    assert len(a.market_orders) == 1
    assert a.market_orders[0]["amount"] == Decimal("0.6")
    assert r.filled == Decimal("1")


def test_cancel_failure_rereads_before_taking():
    """撤单失败时重读订单，按真实成交量决定吃多少，而不是无条件吃全量。"""
    a = _Adapter(fill_sequence=[Decimal("0.7")], cancel_error="撤单失败")
    r = _run(a)
    assert len(a.market_orders) == 1
    assert a.market_orders[0]["amount"] == Decimal("0.3")
    assert "撤单失败" in r.note


def test_fully_filled_after_cancel_failure_places_no_taker():
    """撤单失败但订单其实已全部成交 —— 绝不能再吃单。"""
    a = _Adapter(fill_sequence=[Decimal("1")], cancel_error="撤单失败")
    a._fills = [Decimal("1")]
    r = _run(a)
    assert a.market_orders == []
    assert r.used_taker is False


def test_zero_delta_places_nothing():
    a = _Adapter()
    r = _run(a, delta=Decimal("0"))
    assert a.limit_orders == [] and a.market_orders == []
    assert r.filled == Decimal("0")
```

```python
@dataclass(frozen=True)
class HedgeFillResult:
    """一次对冲下单的结果。

    filled 是最终覆盖掉的数量（maker 成交 + 吃单补足）。
    note 记录异常路径，供排查。
    """

    filled: Decimal
    used_taker: bool
    note: str = ""


async def maker_first_hedge(
    adapter,
    market: str,
    target_delta: Decimal,
    *,
    timeout_s: float = 15.0,
    poll_s: float = 1.0,
) -> HedgeFillResult:
    """先挂 maker，超时未成交则撤单、按剩余量吃单。

    Extended maker 费率为 0、taker 约 3bps。高频调仓下纯吃单的成本会成为
    吃掉本金的主要项，因此优先走 maker。

    成交判定用订单状态而非持仓差值：持仓差值遇到部分成交会误判成「已成交」，
    导致剩余量既没对冲上、挂单又变成无人认领的孤儿单。
    """
    if target_delta == 0:
        return HedgeFillResult(filled=Decimal(0), used_taker=False, note="目标为零")

    side = Side.BUY if target_delta > 0 else Side.SELL
    amount = abs(target_delta)

    # 挂在本方最优价：卖挂 ask、买挂 bid。挂在对手价会立刻成交而被 post_only 拒绝
    quote = await adapter.get_market_price(market)
    price = quote.ask if side is Side.SELL else quote.bid

    order = await adapter.place_limit_order(
        market, side, amount, price, post_only=True
    )
    order_id = getattr(order, "id", None) or (
        order.get("id") if isinstance(order, dict) else None
    )

    async def _filled_qty() -> Decimal:
        got = await adapter.get_order_by_id(market, order_id)
        raw = getattr(got, "filled_qty", None)
        if raw is None and isinstance(got, dict):
            raw = got.get("filled_qty")
        return Decimal(str(raw or 0))

    deadline = time.monotonic() + timeout_s
    filled = Decimal(0)
    while time.monotonic() < deadline:
        await asyncio.sleep(poll_s)
        filled = await _filled_qty()
        if filled >= amount:
            return HedgeFillResult(filled=filled, used_taker=False, note="maker 全部成交")

    note = ""
    try:
        await adapter.cancel_order(market, order_id)
    except Exception as exc:  # noqa: BLE001
        note = f"撤单失败（{exc}）"
        logger.warning("对冲撤单失败，将按重读结果决定是否吃单：%s", exc)

    # 撤单成功与否都重读一次：撤单前的瞬间可能又成交了一部分，
    # 按陈旧数据吃单会超额对冲
    filled = await _filled_qty()
    remaining = amount - filled
    if remaining <= 0:
        return HedgeFillResult(
            filled=filled, used_taker=False,
            note=(note + "；重读发现已全部成交，未吃单").lstrip("；"),
        )

    await adapter.market_order(market, side, remaining)
    return HedgeFillResult(
        filled=amount, used_taker=True,
        note=(note + f"；maker 成交 {filled}，吃单补 {remaining}").lstrip("；"),
    )
```

```python
    async def _rebalance(self, target_hedge: Decimal) -> str:
        """把 hedge 腿调到目标有符号数量。"""
        msg = f"再平衡 hedge → {target_hedge}"
        if self.config.dry_run:
            logger.info("[dry_run] %s（未真正下单）", msg)
            return f"[dry_run] {msg}"

        if self.config.maker_first_timeout_s > 0:
            current = (await self.hedge.get_position(self.config.market)).signed_size
            delta = target_hedge - current
            result = await maker_first_hedge(
                self.hedge,
                self.config.market,
                delta,
                timeout_s=self.config.maker_first_timeout_s,
            )
            logger.info("%s 完成（maker 优先）：%s", msg, result.note)
            return f"{msg}（{result.note}）"

        result = await self.hedge.hedge(self.config.market, target_hedge)
        logger.info("%s 完成：%s", msg, result)
        return msg
```
