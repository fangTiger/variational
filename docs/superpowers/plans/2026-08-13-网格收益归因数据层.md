# 网格收益归因数据层 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 让系统能回答"网格自己到底赚了多少"——把闭环利润变成可跨重启累计、可审计的持久数据，并在 4 周后自动给出去留结论。

**Architecture:** 实盘交易进程只做一件事：每笔成交 append 一行 JSONL（约 10 行改动）。一个独立的旁路进程负责导入 SQLite、离线配对成闭环、算归因、发告警。交易进程零新增故障面，数据库损坏可随时从 JSONL 全量重建。

**Tech Stack:** Python 3.11、SQLite（标准库 `sqlite3`，WAL 模式）、pytest、launchd。

**背景（实施者必读）：**
- 设计文档在 `docs/superpowers/specs/2026-08-11-网格收益归因数据层-design.md`
- 闭环 = 同一格号上一买一卖配成对，赚的价差就是网格的真实利润。maker 费为 0，故毛利即净利
- 引擎里已有配对逻辑（`grid/grid_engine.py:1388-1400`），但结果存在内存 `self._loop_fills`，**每次重启归零**，且跨重启的半个闭环会永久漏计。所以这里改为从完整成交流**离线**重新配对
- 权益快照已有：`data/grid_monitor.jsonl`，每小时一条，由 launchd 任务 `com.variational.grid-monitor` 写入
- 跑测试：`.venv/bin/python -m pytest tests/ -q`，当前基线 288 passed
- 所有注释和文档用中文，标识符用英文

```python
"""成交明细追写。

引擎侧只负责 append 一行纯文本，这是能想到的最低风险写入方式：
不引入数据库依赖、不占锁、失败也只丢一条统计数据。
闭环配对由旁路进程离线做（见 grid/attribution/pairing.py），
因为引擎内的配对依赖内存态 _loop_fills，跨重启必然断裂。
"""

from __future__ import annotations

import json
from decimal import Decimal
from pathlib import Path

from infra.logger import get_logger

logger = get_logger("fill_log")


def build_fill_record(
    *,
    fill_id: str,
    ts: float,
    level: int,
    side: str,
    price: Decimal,
    qty: Decimal,
    engine_run_id: str,
) -> dict:
    """构造一条成交记录。Decimal 转 float 便于 JSON 序列化。"""
    return {
        "fill_id": str(fill_id),
        "ts": float(ts),
        "level": int(level),
        "side": str(side),
        "price": float(price),
        "qty": float(qty),
        "engine_run_id": str(engine_run_id),
    }


def append_fill(path: Path, record: dict) -> None:
    """追加一行。任何异常只记日志，绝不向上抛——交易循环优先。"""
    try:
        path.parent.mkdir(parents=True, exist_ok=True)
        with path.open("a", encoding="utf-8") as f:
            f.write(json.dumps(record, ensure_ascii=False) + "\n")
    except Exception as exc:  # noqa: BLE001
        logger.warning("成交追写失败（不影响交易）：%s", exc)
```

```python
"""归因数据的 SQLite 存储。

选 SQLite 而非 PostgreSQL：4 周数据量约 2000 行、单机单写者。
无服务、文件即数据库、可随时从 JSONL 全量重建。
启用 WAL，让面板和分析脚本能在导入期间并发读。
"""

from __future__ import annotations

import json
import sqlite3
from pathlib import Path

from infra.logger import get_logger

logger = get_logger("attribution")

_SCHEMA = """
CREATE TABLE IF NOT EXISTS fills (
    fill_id       TEXT PRIMARY KEY,
    ts            REAL NOT NULL,
    level         INTEGER NOT NULL,
    side          TEXT NOT NULL,
    price         REAL NOT NULL,
    qty           REAL NOT NULL,
    engine_run_id TEXT
);
CREATE INDEX IF NOT EXISTS idx_fills_level_ts ON fills(level, ts);

CREATE TABLE IF NOT EXISTS closed_loops (
    loop_id     TEXT PRIMARY KEY,
    ts          REAL NOT NULL,
    level       INTEGER NOT NULL,
    buy_price   REAL NOT NULL,
    sell_price  REAL NOT NULL,
    qty         REAL NOT NULL,
    gross_pnl   REAL NOT NULL
);
CREATE INDEX IF NOT EXISTS idx_loops_ts ON closed_loops(ts);

CREATE TABLE IF NOT EXISTS equity_snapshots (
    ts          REAL PRIMARY KEY,
    equity      REAL NOT NULL,
    inv_usd     REAL,
    price       REAL,
    mode        TEXT,
    cash_flow   REAL NOT NULL DEFAULT 0
);

CREATE TABLE IF NOT EXISTS funding (
    funding_id  TEXT PRIMARY KEY,
    ts          REAL NOT NULL,
    fee         REAL NOT NULL
);
"""


def connect(db_path: Path) -> sqlite3.Connection:
    db_path.parent.mkdir(parents=True, exist_ok=True)
    conn = sqlite3.connect(str(db_path))
    conn.execute("PRAGMA journal_mode=WAL")
    return conn


def init_schema(conn: sqlite3.Connection) -> None:
    conn.executescript(_SCHEMA)
    conn.commit()


def _iter_json_lines(path: Path):
    """逐行读 JSONL，跳过损坏行——单行坏了不该让整次导入失败。"""
    if not path.exists():
        return
    for line in path.read_text(encoding="utf-8").splitlines():
        if not line.strip():
            continue
        try:
            yield json.loads(line)
        except Exception:  # noqa: BLE001
            logger.warning("跳过损坏行：%.80s", line)


def import_fills(conn: sqlite3.Connection, src: Path) -> int:
    """导入成交明细，返回新增行数。fill_id 为主键，重复导入自动忽略。"""
    inserted = 0
    for rec in _iter_json_lines(src):
        try:
            cur = conn.execute(
                "INSERT OR IGNORE INTO fills"
                " (fill_id, ts, level, side, price, qty, engine_run_id)"
                " VALUES (?, ?, ?, ?, ?, ?, ?)",
                (
                    str(rec["fill_id"]), float(rec["ts"]), int(rec["level"]),
                    str(rec["side"]), float(rec["price"]), float(rec["qty"]),
                    rec.get("engine_run_id"),
                ),
            )
            inserted += cur.rowcount
        except Exception as exc:  # noqa: BLE001
            logger.warning("跳过无法入库的成交记录：%s", exc)
    conn.commit()
    return inserted
```

```python
"""从成交流离线配对出闭环。纯函数，无 IO，便于测试与重算。

为什么不用引擎内的配对结果：那份逻辑依赖内存态 self._loop_fills，
进程重启即清空，重启前挂着的半个闭环再也配不上，利润永久漏计
（2026-08-13 一天重启 5 次）。离线重算不受此影响，且配对规则
将来要修正时不必动交易进程。
"""

from __future__ import annotations


def pair_fills(fills: list[dict]) -> list[dict]:
    """按格号配对反向成交，返回闭环列表（按时间升序）。

    规则与引擎一致：同格反向即配对，数量取小值，剩余留待下次。
    loop_id 由两个 fill_id 拼成，因此对同一批输入是确定的——
    这保证重复入库不会重复计算利润。
    """
    pending: dict[int, list[dict]] = {}
    loops: list[dict] = []

    for fill in sorted(fills, key=lambda x: (float(x["ts"]), str(x["fill_id"]))):
        level = int(fill["level"])
        side = str(fill["side"]).upper()
        remaining = float(fill["qty"])
        queue = pending.setdefault(level, [])

        # 先和同格反向的挂账逐个配对
        while remaining > 0 and queue and str(queue[0]["side"]).upper() != side:
            head = queue[0]
            matched = min(remaining, float(head["qty"]))
            buy_price = float(head["price"]) if side == "SELL" else float(fill["price"])
            sell_price = float(fill["price"]) if side == "SELL" else float(head["price"])
            loops.append({
                "loop_id": f"{head['fill_id']}+{fill['fill_id']}",
                "ts": float(fill["ts"]),
                "level": level,
                "buy_price": buy_price,
                "sell_price": sell_price,
                "qty": matched,
                "gross_pnl": (sell_price - buy_price) * matched,
            })
            remaining -= matched
            head["qty"] = float(head["qty"]) - matched
            if head["qty"] <= 0:
                queue.pop(0)

        if remaining > 0:
            queue.append({**fill, "qty": remaining})

    return loops
```

def import_equity_snapshots(conn: sqlite3.Connection, src: Path) -> int:
    """导入权益快照（来自 data/grid_monitor.jsonl），返回新增行数。

    权益为 0 的记录是接口超时的产物，必须剔除：它们会把回撤统计
    从真实的 $27 虚增到 $972（2026-08-10 实测）。
    """
    inserted = 0
    for rec in _iter_json_lines(src):
        equity = rec.get("equity") or 0
        if float(equity) <= 0:
            continue
        cur = conn.execute(
            "INSERT OR IGNORE INTO equity_snapshots"
            " (ts, equity, inv_usd, price, mode) VALUES (?, ?, ?, ?, ?)",
            (
                float(rec["ts"]), float(equity),
                rec.get("inv_usd"), rec.get("price"), rec.get("mode"),
            ),
        )
        inserted += cur.rowcount
    conn.commit()
    return inserted


def import_funding(conn: sqlite3.Connection, rows: list[dict]) -> int:
    """导入资金费流水，返回新增行数。"""
    inserted = 0
    for rec in rows:
        cur = conn.execute(
            "INSERT OR IGNORE INTO funding (funding_id, ts, fee) VALUES (?, ?, ?)",
            (str(rec["funding_id"]), float(rec["ts"]), float(rec["fee"])),
        )
        inserted += cur.rowcount
    conn.commit()
    return inserted

def fetch_funding(market: str = "BTC-USD", limit: int = 500) -> list[dict]:
    """返回归一化的资金费记录；失败返回空列表，不抛错。"""
    api_key = os.getenv("X10_GRID_API_KEY")
    if not api_key:
        return []
    url = f"{_BASE}/api/v1/user/funding/history?market={market}&limit={limit}"
    req = urllib.request.Request(
        url,
        headers={
            "X-Api-Key": api_key,
            "User-Agent": "grid-bot/1.0",
            "Accept": "application/json",
        },
    )
    ctx = ssl.create_default_context(cafile=certifi.where())
    try:
        with urllib.request.urlopen(req, timeout=25, context=ctx) as resp:
            payload = json.loads(resp.read())
    except Exception:  # noqa: BLE001
        return []
    out = []
    for row in payload.get("data") or []:
        try:
            out.append({
                "funding_id": str(row["id"]),
                "ts": float(row["paidTime"]) / 1000.0,
                "fee": float(row["fundingFee"]),
            })
        except Exception:  # noqa: BLE001
            continue
    return out

```python
"""归因计算与去留判据。纯函数，无 IO。"""

from __future__ import annotations

# 残差超过当日闭环利润这个比例，就认为归因有缺口
_RESIDUAL_TOLERANCE_FRAC = 0.20
# 残差绝对值下限：闭环利润接近 0 时，比例判据会过于敏感
_RESIDUAL_TOLERANCE_ABS = 1.0

OBSERVATION_DAYS = 28.0
MIN_GRID_ANNUALISED_PCT = 15.0
MAX_DRAWDOWN_PCT = 12.0


def compute_attribution(
    *,
    equity_start: float,
    equity_end: float,
    loops: list[dict],
    funding_total: float,
    cash_flow: float,
    unrealised_change: float | None = None,
    days: float = 1.0,
) -> dict:
    """把权益变动拆成网格闭环利润与方向性盈亏。

    恒等式：权益变动 ≡ 闭环利润 + 未实现盈亏变动 + 资金费 + 出入金
    出入金先从权益变动里剔除，否则一笔入金会被算成收益。
    """
    equity_change = equity_end - equity_start - cash_flow
    grid_pnl = sum(float(loop["gross_pnl"]) for loop in loops)
    directional_pnl = equity_change - grid_pnl - funding_total

    residual = 0.0
    if unrealised_change is not None:
        expected = grid_pnl + unrealised_change + funding_total
        residual = equity_change - expected

    tolerance = max(abs(grid_pnl) * _RESIDUAL_TOLERANCE_FRAC, _RESIDUAL_TOLERANCE_ABS)
    principal = equity_start if equity_start > 0 else 1.0
    grid_annualised_pct = (grid_pnl / max(days, 1e-9)) * 365.0 / principal * 100.0

    return {
        "equity_change": equity_change,
        "grid_pnl": grid_pnl,
        "directional_pnl": directional_pnl,
        "funding_total": funding_total,
        "residual": residual,
        "has_gap": abs(residual) > tolerance,
        "grid_annualised_pct": grid_annualised_pct,
        "days": days,
    }


def verdict(
    *,
    grid_annualised_pct: float,
    equity_annualised_pct: float,
    max_drawdown_pct: float,
    has_gap: bool,
    days: float,
) -> dict:
    """4 周去留判据。三方评审一致：闭环年化只是必要不充分条件。"""
    if days < OBSERVATION_DAYS:
        return {
            "decided": False,
            "should_stop": False,
            "reason": f"观察期未满（{days:.1f}/{OBSERVATION_DAYS:.0f} 天）",
        }
    reasons = []
    if grid_annualised_pct < MIN_GRID_ANNUALISED_PCT:
        reasons.append(
            f"闭环年化 {grid_annualised_pct:.1f}% < {MIN_GRID_ANNUALISED_PCT:.0f}%"
        )
    if equity_annualised_pct <= 0:
        reasons.append(f"净值年化 {equity_annualised_pct:.1f}% ≤ 0")
    if max_drawdown_pct >= MAX_DRAWDOWN_PCT:
        reasons.append(
            f"最大回撤 {max_drawdown_pct:.1f}% ≥ {MAX_DRAWDOWN_PCT:.0f}%"
        )
    if has_gap:
        reasons.append("归因存在缺口，数据不可信")
    return {
        "decided": True,
        "should_stop": bool(reasons),
        "reason": "；".join(reasons) if reasons else "全部判据通过",
    }
```

"""归因主流程：导入 → 配对 → 计算 → 落盘。launchd 每小时调用。

全部只读交易所、只写本地 SQLite，不碰账户、不下单。
"""

from __future__ import annotations

from infra.runtime import ensure_ssl_cert

ensure_ssl_cert()

import argparse  # noqa: E402
import json  # noqa: E402
from pathlib import Path  # noqa: E402

from dotenv import load_dotenv  # noqa: E402

from grid.attribution.pairing import pair_fills  # noqa: E402
from grid.attribution.report import compute_attribution, verdict  # noqa: E402
from grid.attribution.store import (  # noqa: E402
    connect,
    import_equity_snapshots,
    import_fills,
    import_funding,
    init_schema,
    load_fills,
    save_loops,
)
from tools.fetch_funding import fetch_funding  # noqa: E402

_ROOT = Path(__file__).resolve().parent.parent
_DATA = _ROOT / "data"
_DB = _DATA / "grid.db"
_RESULT = _DATA / "attribution.json"


def run() -> dict:
    load_dotenv(str(_ROOT / ".env"))
    conn = connect(_DB)
    init_schema(conn)

    import_fills(conn, _DATA / "fills.jsonl")
    import_equity_snapshots(conn, _DATA / "grid_monitor.jsonl")
    import_funding(conn, fetch_funding())

    loops = pair_fills(load_fills(conn))
    save_loops(conn, loops)

    snaps = conn.execute(
        "SELECT ts, equity FROM equity_snapshots ORDER BY ts"
    ).fetchall()
    funding_total = conn.execute(
        "SELECT COALESCE(SUM(fee), 0) FROM funding"
    ).fetchone()[0]

    if len(snaps) < 2:
        result = {"decided": False, "reason": "权益快照不足，无法归因"}
        _RESULT.write_text(json.dumps(result, ensure_ascii=False), encoding="utf-8")
        return result

    days = (snaps[-1][0] - snaps[0][0]) / 86400.0
    equities = [s[1] for s in snaps]
    peak = equities[0]
    max_dd = 0.0
    for eq in equities:
        peak = max(peak, eq)
        max_dd = max(max_dd, (peak - eq) / peak * 100.0)

    attribution = compute_attribution(
        equity_start=equities[0],
        equity_end=equities[-1],
        loops=[{"gross_pnl": row[0]} for row in
               conn.execute("SELECT gross_pnl FROM closed_loops").fetchall()],
        funding_total=float(funding_total),
        cash_flow=0.0,
        days=days,
    )
    equity_annualised = (
        attribution["equity_change"] / max(days, 1e-9) * 365.0 / equities[0] * 100.0
    )
    decision = verdict(
        grid_annualised_pct=attribution["grid_annualised_pct"],
        equity_annualised_pct=equity_annualised,
        max_drawdown_pct=max_dd,
        has_gap=attribution["has_gap"],
        days=days,
    )
    result = {
        **attribution,
        "equity_annualised_pct": equity_annualised,
        "max_drawdown_pct": max_dd,
        "loops_count": len(loops),
        **decision,
    }
    _RESULT.write_text(
        json.dumps(result, ensure_ascii=False, indent=2), encoding="utf-8"
    )
    return result


def main() -> None:
    parser = argparse.ArgumentParser(description="网格收益归因")
    parser.add_argument("--report", action="store_true", help="打印结果")
    args = parser.parse_args()
    result = run()
    if args.report or True:
        print(
            f"闭环 {result.get('loops_count', 0)} 个 | "
            f"网格 ${result.get('grid_pnl', 0):+.2f} | "
            f"方向性 ${result.get('directional_pnl', 0):+.2f} | "
            f"闭环年化 {result.get('grid_annualised_pct', 0):.1f}% | "
            f"{result.get('reason', '')}"
        )


if __name__ == "__main__":
    main()

**背景：** 熔断的峰值权益目前不区分资金流与行情亏损，出入金后需要人工删除 `data/equity_peak.json`。曾用"权益跳变 ≥8% 判为出入金"自动处理，但真实快速亏损同样 ≥8%，导致熔断可被绕过，已移除（见 `grid/grid_engine.py` 中的说明注释）。正解是读交易所真实流水。端点 `/api/v1/user/assetOperations` 已验证返回 200。

- [ ] **Step 1: 在 `tools/fetch_funding.py` 末尾加取流水函数**

```python
def fetch_asset_operations(limit: int = 50) -> list[dict]:
    """拉取出入金流水。失败返回空列表——取不到就不调整基准，宁可漏调不可误调。

    返回 [{"ts": 秒级时间戳, "amount": 有符号金额}]，入金为正、出金为负。
    """
    api_key = os.getenv("X10_GRID_API_KEY")
    if not api_key:
        return []
    url = f"{_BASE}/api/v1/user/assetOperations?limit={limit}"
    req = urllib.request.Request(
        url,
        headers={
            "X-Api-Key": api_key,
            "User-Agent": "grid-bot/1.0",
            "Accept": "application/json",
        },
    )
    ctx = ssl.create_default_context(cafile=certifi.where())
    try:
        with urllib.request.urlopen(req, timeout=25, context=ctx) as resp:
            payload = json.loads(resp.read())
    except Exception:  # noqa: BLE001
        return []
    out = []
    for row in payload.get("data") or []:
        try:
            amount = float(row.get("amount"))
            op_type = str(row.get("type", "")).upper()
            if "WITHDRAW" in op_type and amount > 0:
                amount = -amount
            ts_raw = row.get("createdTime") or row.get("time") or 0
            out.append({"ts": float(ts_raw) / 1000.0, "amount": amount})
        except Exception:  # noqa: BLE001
            continue
    return out
```

```python
    @staticmethod
    def adjust_peak_for_cash_flow(
        *,
        peak: float,
        operations: list[dict],
        since_ts: float,
    ) -> float:
        """按真实出入金流水调整峰值基准。

        用交易所流水而非猜权益跳变：曾用"跳变 ≥8% 判为出入金"，
        但真实快速亏损同样 ≥8%，会让熔断被绕过（实测 1000→900→810
        两次 10% 亏损逐次重置峰值、永不触发）。
        """
        delta = sum(
            float(op["amount"])
            for op in operations
            if float(op.get("ts") or 0) > since_ts
        )
        return max(peak + delta, 0.0)
```

"""失败轮次日志聚合测试。

2026-08-13 DNS 故障 3 小时打出 3686 个连接错误 + 完整 traceback，
日志从 780KB/天涨到 19MB，真实问题被淹没。
同类错误连续出现时应折叠，但第一条和恢复必须可见。
"""

from __future__ import annotations

from types import SimpleNamespace

from grid.grid_engine import _FAILURE_LOG_EVERY, GridEngine


def _eng():
    return SimpleNamespace(_failure_streak={"signature": None, "count": 0})


def test_first_failure_is_logged():
    eng = _eng()
    assert GridEngine._should_log_failure(eng, "ClientConnectorDNSError") is True


def test_repeated_same_failure_is_folded():
    eng = _eng()
    GridEngine._should_log_failure(eng, "DNS")
    for _ in range(_FAILURE_LOG_EVERY - 2):
        assert GridEngine._should_log_failure(eng, "DNS") is False


def test_periodic_summary_still_emitted():
    """完全静默也不行——长时间故障要留下心跳。"""
    eng = _eng()
    logged = sum(
        1 for _ in range(_FAILURE_LOG_EVERY * 3)
        if GridEngine._should_log_failure(eng, "DNS")
    )
    assert logged == 3

    """换了一种错误必须立刻可见，不能被上一种的计数折叠掉。"""
    eng = _eng()
    GridEngine._should_log_failure(eng, "DNS")
    assert GridEngine._should_log_failure(eng, "TimeoutError") is True
