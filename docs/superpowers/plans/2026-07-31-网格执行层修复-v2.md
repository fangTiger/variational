## 一、v1 被推翻的经过（必读，避免重蹈覆辙）

v1 计划的核心论据是"222 笔消失订单中 79 笔（35%）判不出终态"。**这个数字是错的。**

清洗测试桩数据（`o1` / `id=new-*` / 价格 88、100）后：

| 指标 | v1 报告 | 真实值 |
|---|---|---|
| 终态未知 | 79 | **15** |
| 真实翻单 | 143 | **19** |
| 日均翻单 | ~36/天 | **4.75/天** |

而那 15 笔真实未知的根因链是：

```
TPSL tick 取整 bug（容差 0.1 < 半个 tick 0.5，数学上永不匹配）
  └→ 每 34 秒重挂一次 TPSL，07-29 单日 4423 条 TPSL 日志
      └→ 订单历史 limit=100 只覆盖约 56 分钟
          └→ 存活 67~1676 分钟（中位数 360 分钟）的老 LIMIT 单落在窗口外
              └→ 查不到终态
```

**三条已核实的结论：**

1. 15 笔真实未知全部发生在 07-28 21:46 ~ 07-30 09:46；按 ID 兜底是 `9be3174`（07-30 16:34）加入的，**晚于全部事件**
2. 当前历史查询已带 `order_type="LIMIT"` 过滤，TPSL 洪水进不来
3. Codex 已交付的 tick 修复斩断了这条链的源头

**所以 v1 的 Task 2（`find_order_in_history`）整个作废——它要修的问题已经修好了。**

**教训**：测试往 `log/grid_engine_*.log` 写假数据，直接毁掉了一次事故诊断。Task 1 修它。

"""挂单去重与翻单重试测试。

背景：翻单失败即永久丢失止盈腿；同档重复挂单产生孤儿单与库存失真。
"""
from __future__ import annotations

import asyncio
from types import SimpleNamespace

from adapters.base import Side
from grid.grid_engine import GridConfig, GridEngine


class FlakyExt:
    """桩：前 N 次挂单抛异常，之后成功。"""

    def __init__(self, fail_times=0):
        self.fail_times = fail_times
        self.placed: list[dict] = []

    async def get_open_orders(self, market):
        return []

    async def get_orders_history(self, market, limit=100, **kwargs):
        return []

    async def place_limit_order(self, market, side, amount, price, **kw):
        if self.fail_times > 0:
            self.fail_times -= 1
            raise RuntimeError("模拟网络故障")
        self.placed.append({"side": side, "price": float(price)})
        return SimpleNamespace(data=SimpleNamespace(id=f"ok-{len(self.placed)}"))


def test_duplicate_level_is_skipped() -> None:
    """同一档位已有跟踪单时不得重复挂。"""
    ext = FlakyExt()
    eng = GridEngine(ext, GridConfig(dry_run=False))
    eng._orders = {558: {"id": "existing", "side": Side.BUY}}

    asyncio.run(eng._place(558, Side.BUY, 0.0, why="测试"))

    assert not ext.placed, "同档不得重复挂单"
    assert eng._orders[558]["id"] == "existing", "既有跟踪记录不得被覆盖"


def test_failed_placement_is_retried_next_round() -> None:
    """挂单失败要进重试队列，下一轮重试成功。"""
    ext = FlakyExt(fail_times=1)
    eng = GridEngine(ext, GridConfig(dry_run=False))

    asyncio.run(eng._place(558, Side.BUY, 0.0, why="测试"))
    assert not ext.placed, "首次应失败"
    assert 558 in eng._retry, "失败必须进重试队列"

    asyncio.run(eng._handle_fills(0.0))       # 下一轮：队列被 drain
    assert len(ext.placed) == 1, "重试必须成功挂出"
    assert 558 not in eng._retry, "成功后必须出队"


def test_retry_respects_blocked_side() -> None:
    """冻结方向的重试必须跳过，不得绕过风控。"""
    ext = FlakyExt(fail_times=1)
    eng = GridEngine(ext, GridConfig(dry_run=False))
    asyncio.run(eng._place(558, Side.BUY, 0.0, why="测试"))

    asyncio.run(eng._handle_fills(0.0, blocked_side="BUY"))

    assert not ext.placed, "冻结方向不得重试挂出"
    assert 558 in eng._retry, "被冻结时保留在队列里等解冻"
```

### Task 4：循环超时兜底 + 轮询提速

**问题**：x10 SDK 默认 `request_timeout_seconds=500`（`.venv/lib/python3.11/site-packages/x10/config.py:13`），而主循环是 `await run_once()` 跑完才 sleep（`grid_engine.py:1242`）。一个卡住的请求能让成交处理、TPSL、硬止损**同时停摆 8 分钟**，launchd 还认为进程健康。

**最简解**：不逐个改 SDK 超时，直接给整轮加超时。

**Files:** Modify `grid/grid_engine.py`（`run_forever`）、`tools/run_grid.py`

def test_slow_round_is_cancelled_and_loop_continues() -> None:
    """单轮超时必须被取消，循环继续，不得卡死。"""
    import asyncio

    from grid.grid_engine import GridConfig, GridEngine

    eng = GridEngine(object(), GridConfig(dry_run=True, round_timeout=0.05))
    calls = {"n": 0}

    async def hang():
        calls["n"] += 1
        await asyncio.sleep(10)

    eng.run_once = hang

    async def drive():
        task = asyncio.create_task(eng.run_forever())
        await asyncio.sleep(0.3)
        task.cancel()

    asyncio.run(drive())
    assert calls["n"] >= 2, "超时后必须继续下一轮，而不是卡死在第一轮"
```

            start = time.monotonic()
            try:
                await asyncio.wait_for(self.run_once(), timeout=self.config.round_timeout)
                self._consecutive_failures = 0
            except asyncio.TimeoutError:
                self._consecutive_failures += 1
                logger.error(
                    "本轮超过 %.0fs 未完成已取消（连续失败 %d 次）",
                    self.config.round_timeout,
                    self._consecutive_failures,

            except Exception:  # noqa: BLE001
                self._consecutive_failures += 1
                logger.exception("本轮异常（连续失败 %d 次）", self._consecutive_failures)
            if self._consecutive_failures >= 5:
                logger.critical("已连续 %d 轮失败，请人工检查", self._consecutive_failures)
            logger.debug("本轮耗时 %.1fs", time.monotonic() - start)
